YouTube - Profiles - collect by URL


import requests
import json

headers = {
    "Authorization": "Bearer fb28f17b77e57fa811bf58f015ca9d05f9415acfb985fc3ad3d6c18a45abb970",
    "Content-Type": "application/json",
}

data = json.dumps({
    "input": [{"url":"https://www.youtube.com/@MrBeast/about"},{"url":"https://www.youtube.com/@jaidenanimations/about"}],
})

response = requests.post(
    "https://api.brightdata.com/datasets/v3/scrape?dataset_id=gd_lk538t2k2p1k3oos71&notify=false&include_errors=true",
    headers=headers,
    data=data
)

print(response.json())
