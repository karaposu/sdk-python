TikTok - Posts by Profile Fast API - collect by URL


import requests
import json

headers = {
    "Authorization": "Bearer fb28f17b77e57fa811bf58f015ca9d05f9415acfb985fc3ad3d6c18a45abb970",
    "Content-Type": "application/json",
}

data = json.dumps({
    "input": [{"url":"https://www.tiktok.com/@bbc"},{"url":"https://www.tiktok.com/@portalotempo"}],
})

response = requests.post(
    "https://api.brightdata.com/datasets/v3/scrape?dataset_id=gd_m7n5v2gq296pex2f5m&notify=false&include_errors=true",
    headers=headers,
    data=data
)

print(response.json())
