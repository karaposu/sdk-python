import requests
import json

headers = {
    "Authorization": "Bearer fb28f17b77e57fa811bf58f015ca9d05f9415acfb985fc3ad3d6c18a45abb970",
    "Content-Type": "application/json",
}

data = json.dumps({
    "input": [{"url":"https://www.tiktok.com/@heymrcat/video/7216019547806092550"},{"url":"https://www.tiktok.com/@mmeowmmia/video/7077929908365823237"}],
})

response = requests.post(
    "https://api.brightdata.com/datasets/v3/scrape?dataset_id=gd_lkf2st302ap89utw5k&notify=false&include_errors=true",
    headers=headers,
    data=data
)

print(response.json())
