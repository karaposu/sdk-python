import requests
import json

headers = {
    "Authorization": "Bearer fb28f17b77e57fa811bf58f015ca9d05f9415acfb985fc3ad3d6c18a45abb970",
    "Content-Type": "application/json",
}

data = json.dumps({
    "input": [{"url":"https://www.tiktok.com/@bellapoarch/video/7433494424040017194","country":""}],
})

response = requests.post(
    "https://api.brightdata.com/datasets/v3/scrape?dataset_id=gd_lu702nij2f790tmv9h&notify=false&include_errors=true",
    headers=headers,
    data=data
)

print(response.json())
