TikTok - Profiles - discover by search url


import requests
import json

headers = {
    "Authorization": "Bearer fb28f17b77e57fa811bf58f015ca9d05f9415acfb985fc3ad3d6c18a45abb970",
    "Content-Type": "application/json",
}

data = json.dumps({
    "input": [{"search_url":"https://www.tiktok.com/explore?lang=en","country":"US"},{"search_url":"https://www.tiktok.com/search?lang=en&q=music&t=1685628060008","country":"FR"}],
})

response = requests.post(
    "https://api.brightdata.com/datasets/v3/scrape?dataset_id=gd_l1villgoiiidt09ci&notify=false&include_errors=true&type=discover_new&discover_by=search_url",
    headers=headers,
    data=data
)

print(response.json())
