import requests
import json

headers = {
    "Authorization": "Bearer fb28f17b77e57fa811bf58f015ca9d05f9415acfb985fc3ad3d6c18a45abb970",
    "Content-Type": "application/json",
}

data = json.dumps({
    "input": [{"url":"https://www.tiktok.com/discover/dog1"},{"url":"https://www.tiktok.com/channel/anime"},{"url":"https://www.tiktok.com/music/Nirvana-Steeve-West-Remix-7166220356133324802"},{"url":"https://www.tiktok.com/explore?lang=en"}],
})

response = requests.post(
    "https://api.brightdata.com/datasets/v3/scrape?dataset_id=gd_m736hjp71lejc5dc0l&notify=false&include_errors=true",
    headers=headers,
    data=data
)

print(response.json())
